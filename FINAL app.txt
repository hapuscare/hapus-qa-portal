def send_expiry_alerts(to_email,to_whatsapp,worker_name,training_name,expiry_date,days_left):
    results=[]
    try:
        sender=st.secrets["email"]["sender"]; pwd=st.secrets["email"]["password"]
        server=st.secrets["email"]["smtp_server"]; port=st.secrets["email"]["smtp_port"]
        manager_email=st.secrets["manager"]["email"]
        subject=f"🚨 {days_left}d left - {worker_name} - {training_name}"
        body=f"Hi {worker_name}\nTraining: {training_name}\nExpiry: {expiry_date}\nDays left: {days_left}\n{'🔴 BLOCKED FROM ROTA per CIW Reg 35' if days_left<=7 else 'Book renewal'}\n\n{SAFE_FOOTER}"
        msg=MIMEMultipart(); msg['From']=sender; msg['To']=to_email; msg['Subject']=subject
        msg.attach(MIMEText(body,'plain'))
        with smtplib.SMTP(server,port) as s:
            s.starttls(); s.login(sender,pwd)
            s.sendmail(sender,[to_email,manager_email],msg.as_string())
        results.append(f"✅ Email to {to_email}+manager")
    except Exception as e:
        results.append(f"❌ Email: {e}")
    if TWILIO_AVAILABLE:
        try:
            sid=st.secrets["twilio"]["account_sid"]; token=st.secrets["twilio"]["auth_token"]; from_w=st.secrets["twilio"]["whatsapp_from"]
            client=Client(sid,token)
            wa_body=f"🚨 *Hapus Care*\nHi {worker_name}\n*{training_name}* expires in *{days_left}d* ({expiry_date})\n{'🔴 BLOCKED from rota' if days_left<=7 else 'Book renewal'}"
            if to_whatsapp:
                client.messages.create(body=wa_body,from_=from_w,to=to_whatsapp)
                results.append(f"✅ WhatsApp to {to_whatsapp}")
        except Exception as e:
            results.append(f"❌ WhatsApp: {e}")
    return results
